{# Badges on list pages: gold slides tag for sessions with a deck, red skull for a session where a player character died (even if brought back), party tag for one where a character joined, came back or left, and a thumbnail for wiki pages with a picture. search.html and map.html draw the same badges in script. #} {% macro slides_icon() %}{% endmacro %} {% macro picture_icon() %}{% endmacro %} {% macro slides_tag(post) %}{% if post.slide_count %}{{ slides_icon() }}{{ post.slide_count }} slide{{ "s" if post.slide_count != 1 }}{% endif %}{% endmacro %} {% macro pic_thumb(page) %}{% if page.thumb %}{% set n = page.pictures | length %}{% if n > 1 %}{{ picture_icon() }}{{ n }}{% endif %}{% endif %}{% endmacro %} {% macro skull_icon() %}{% endmacro %} {% macro party_icon() %}{% endmacro %} {% macro death_tag(post) %}{% if post.deaths %}{{ skull_icon() }}{{ post.death_names }}{% endif %}{% endmacro %} {% macro party_tag(post) %}{% if post.changes %}{{ party_icon() }}{{ post.party_label }}{% endif %}{% endmacro %} {# Slides, skull and party badges after a meta line such as "12 min read", each after a middle dot. #} {% macro meta_tags(post) %}{% if post.slide_count %} · {{ slides_tag(post) }}{% endif %}{% if post.deaths %} · {{ death_tag(post) }}{% endif %}{% if post.changes %} · {{ party_tag(post) }}{% endif %}{% endmacro %} {# The same badges in a session list row, after the date. #} {% macro row_tags(post) %}{% if post.slide_count or post.deaths or post.changes %}{{ slides_tag(post) }} {{ death_tag(post) }} {{ party_tag(post) }}{% endif %}{% endmacro %}
